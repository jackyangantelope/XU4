cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD adafruit_fruit_jam CACHE STRING "Public target" FORCE)
set(PICO_PSRAM_CS_PIN 47 CACHE STRING "PSRAM CS" FORCE)
set(PICO_PSRAM_SIZE_BYTES 0x800000 CACHE STRING "PSRAM size" FORCE)
set(PICO_STACK_SIZE 32768)
set(PICO_CORE1_STACK_SIZE 4096)
if(NOT DEFINED PICO_SDK_PATH OR NOT DEFINED PICO_PIO_USB_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH and PICO_PIO_USB_PATH")
endif()
set(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
set(PICO_CXX_ENABLE_EXCEPTIONS 1)
set(PICO_CXX_ENABLE_RTTI 1)
project(picoXu4Native C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
# Reuse frozen, unmodified libraries; this working copy contains only xu4 sources.
set(LIBS "${PORT_ROOT}/deps/libs")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
add_compile_options(-O2 -ffunction-sections -fdata-sections
  "-ffile-prefix-map=${PORT_ROOT}=."
  "-ffile-prefix-map=${PICO_SDK_PATH}=pico-sdk"
  "-ffile-prefix-map=${PICO_PIO_USB_PATH}=Pico-PIO-USB")
foreach(opt C14N CATALOG DEBUG DOCB FTP HTML HTTP ICONV ICU LZMA MODULES PROGRAMS PYTHON READER SCHEMAS SCHEMATRON TESTS THREADS WRITER ZLIB)
  set(LIBXML2_WITH_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(HAVE_MMAP 0 CACHE INTERNAL "No mmap on bare metal" FORCE)
set(HAVE_MUNMAP 0 CACHE INTERNAL "No mmap on bare metal" FORCE)
set(HAVE_PTHREAD_H 0 CACHE INTERNAL "Newlib header does not implement pthreads" FORCE)
set(HAVE_DLOPEN 0 CACHE INTERNAL "No shared objects on bare metal" FORCE)
set(HAVE_SHLLOAD 0 CACHE INTERNAL "No shared objects on bare metal" FORCE)
add_subdirectory("${LIBS}/libxml2" xml EXCLUDE_FROM_ALL)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/zlib" zlib EXCLUDE_FROM_ALL)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR "${LIBS}/zlib" CACHE PATH "" FORCE)
target_include_directories(zlibstatic PUBLIC "${LIBS}/zlib" "${CMAKE_BINARY_DIR}/zlib")
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(PNG_ARM_NEON off CACHE STRING "" FORCE)
set(PNG_HARDWARE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
set(PNG_BUILD_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/libpng" png EXCLUDE_FROM_ALL)
target_link_libraries(png_static PUBLIC zlibstatic)
target_include_directories(png_static PUBLIC "${CMAKE_BINARY_DIR}/png")
set(LIBXMP_DISABLE_DEPACKERS ON CACHE BOOL "" FORCE)
set(LIBXMP_DISABLE_PROWIZARD ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBS}/libxmp" xmp EXCLUDE_FROM_ALL)
include("platform/native/xu4_sources.cmake")
target_compile_definitions(xu4_core PRIVATE XU4_PICO=1)
target_link_libraries(xu4_core PUBLIC LibXml2 png_static xmp_static pico_stdlib)
set(JACK_WAVE1_HDMI_AUDIO ON)
set(JACK_WAVE1_CONTINUOUS_SILENCE ON)
add_subdirectory("platform/runtime" runtime)
target_compile_definitions(jack_wave1_runtime PRIVATE JACK_WAVE1_USB_NONBLOCKING=1 JACK_WAVE1_MAX_FILES=1)
target_compile_definitions(jack_wave1_runtime PRIVATE JACK_WAVE1_QUIET_INPUT_LOG=1)
target_compile_definitions(jack_wave1_runtime PUBLIC JACK_WAVE1_CONTINUOUS_SILENCE=1)
add_executable(picoXu4Native "platform/native/board_hal.cpp" "platform/native/board_files.c")
target_sources(picoXu4Native PRIVATE "platform/native/xu4_timeline.cpp" "platform/native/xu4_fault.c" "platform/native/xu4_fault_entry.S" "platform/native/xu4_close_diag.c" "platform/native/xu4_integrity.c" "platform/native/xu4_read_diag.c" "platform/native/xu4_boot_diag.cpp" "platform/native/xu4_profile.cpp" "platform/native/xu4_tail_diag.cpp")
target_compile_definitions(picoXu4Native PRIVATE XU4_PICO=1)
target_sources(picoXu4Native PRIVATE "platform/native/native_game_menu.cpp")

target_link_libraries(picoXu4Native PRIVATE xu4_core jack_wave1_runtime hardware_psram)
target_compile_definitions(picoXu4Native PRIVATE PICO_STACK_SIZE=32768 PICO_CORE1_STACK_SIZE=4096 PICO_USE_STACK_GUARDS=1)
pico_add_linker_script_override_path(picoXu4Native "${PORT_ROOT}/cmake/linker")
target_link_options(picoXu4Native PRIVATE -Wl,--gc-sections -Wl,--print-memory-usage)
pico_enable_stdio_uart(picoXu4Native 1)
pico_enable_stdio_usb(picoXu4Native 0)
pico_set_program_name(picoXu4Native "picoXu4Native")
pico_set_program_version(picoXu4Native "0.1.3-public")
include("${PORT_ROOT}/cmake/LoaderPartition.cmake")
xu4_link_for_loader(picoXu4Native)
pico_add_extra_outputs(picoXu4Native)

set_source_files_properties("platform/native/xu4_fault.c" PROPERTIES COMPILE_OPTIONS "-mgeneral-regs-only")


# Keep SDK allocation/mutex behavior; instrument only its existing free wrapper.
get_target_property(xu4_malloc_sources pico_malloc INTERFACE_SOURCES)
list(FILTER xu4_malloc_sources EXCLUDE REGEX "/pico_malloc/malloc[.]c$")
list(APPEND xu4_malloc_sources "${PORT_ROOT}/platform/native/xu4_sdk_malloc_diag.c")
set_property(TARGET pico_malloc PROPERTY INTERFACE_SOURCES "${xu4_malloc_sources}")
